// Criação de coleções


// Criação da coleção player


db.players.aggregate([
  {
    // Stage 1: Deduplicate and Group
    $group: {
      _id: "$LinkPlayer",
      Name: { $first: "$PlayerName" },
      Born: { $first: "$Born" },
      Height: { $first: "$Height" },
      Hand: { $first: "$Hand" },
      PlayerLink: { $first: "$LinkPlayer" }
    }
  },
  {
    // Stage 2: Clean and Format
    $project: {
      _id: 0,
      PlayerLink: {
        $replaceAll: {
          input: "$PlayerLink",
          find: ":",
          replacement: ""
        }
      },
      Name: "$Name",
      Born: "$Born",
      Height: "$Height",
      Hand: "$Hand"
    }
  },
  {
    // Stage 3: Output to new collection
    $merge: "player"
  }
]);



// Criação da coleção tournaments



db.players.aggregate([
  {
    // Stage 1: Criação do campo de agrupamento (Filter)
    $project: {
      Filter: { $concat: ["$Tournament", "", "$Date"] }, // Combina Nome e Data
      Name: "$Tournament",
      Location: "$Location",
      Date: "$Date",
      Ground: "$Ground",
      Prize: "$Prize"
    }
  },
  {
    // Stage 2: Deduplicação e Agrupamento
    $group: {
      _id: "$Filter",
      Name: { $first: "$Name" },
      Location: { $first: "$Location" },
      Date: { $first: "$Date" },
      Ground: { $first: "$Ground" },
      Prize: { $first: "$Prize" }
    }
  },
  {
    // Stage 3: Limpeza Final
    $project: {
      '_id': 0 // Remove o campo _id (que contém o "Filter" combinado)
    }
  },
  {
    // Stage 4: Saída para nova coleção
    $merge: "tournaments"
  }
]);

// Criação de player_borncountry


db.player.aggregate([
  {
    // 1. Criação das colunas iniciais
    $project: {
      pl_city_country: "$Born",
      Country: {
        $arrayElemAt: [
          { $split: ["$Born", ","] }, // Divide a string pela vírgula
          -1                          // Seleciona o último elemento (o país)
        ]
      }
    }
  },
  {
    // 2. Agrupamento para remover duplicados
    $group: {
      _id: {
        pl_city_country: "$pl_city_country",
        Country: "$Country"
      }
    }
  },
  {
    // 3. Reorganização para remover o aninhamento do _id
    $project: {
      _id: 0,
      pl_city_country: "$_id.pl_city_country",
      Country: "$_id.Country"
    }
  },
  {
    // 4. Exportação para a nova coleção
    $merge: "player_borncountry"
  }
])

          $cond: [

            "$Player",

          ]
        },

          $cond: [
            { $eq: ["$WL", "W"] },
            "$Oponent",
            "$Player"
          ]
        }
      }
    }
  ]
);



// Remover jogos duplicados. Entre os duplicados remover aqueles que o resultado é loss (WL = "L").



const losersToDelete = db.games2.aggregate([
  {
    $group: {
      _id: {
        Date: "$Date",
        Tounament: "$Tounament",
        Ground: "$Ground",
        GameRound: "$GameRound",
        winner: "$winner",
        loser: "$loser"
      },
      docs: { $push: { _id: "$_id", WL: "$WL" } },
      count: { $sum: 1 }
    }
  },
  // Filtra apenas os grupos duplicados (count > 1)
  {
    $match: { count: { $gt: 1 } }
  },
  // Extrai apenas os _id dos documentos WL = "L"
  {
    $project: {
      loserIds: {
        $map: {
          input: {
            $filter: {
              input: "$docs",
              as: "d",
              cond: { $eq: ["$$d.WL", "L"] }
            }
          },
          as: "l",
          in: "$$l._id"
        }
      }
    }
  },
  { $unwind: "$loserIds" },
  { $project: { _id: "$loserIds" } }
]).toArray();

// 2) Deletar todos os documentos identificados
if (losersToDelete.length > 0) {
  const ids = losersToDelete.map(d => d._id);
  const res = db.games2.deleteMany({ _id: { $in: ids } });
  print("Removidos:", res.deletedCount);
} else {
  print("Nenhum WL:L encontrado nos grupos duplicados.");
}
